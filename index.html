const alunos = [];

function criar(nome) {
  alunos.push({ id: Date.now(), nome });
  render();
}

function deletar(id) {
  const i = alunos.findIndex(a => a.id === id);
  alunos.splice(i, 1);
  render();
}

async function salvar() {
  await fetch("/api/alunos", {
    method: "POST",
    body: JSON.stringify(alunos),
  });

}
